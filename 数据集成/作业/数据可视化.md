### 第一步：先把数据提取成静态 JSON

在服务器上运行这个脚本，把 `.jsonl` 数据预处理成 HTML 能直接读取的 JSON：

```bash
python3 ~/viz/prepare_data.py
```

**`~/viz/prepare_data.py`** 内容如下：

```python
#!/usr/bin/env python3
import json, os, glob

DATA_DIR = os.path.expanduser("~/data/data")
OUT_DIR  = os.path.expanduser("~/viz/static")
os.makedirs(OUT_DIR, exist_ok=True)

# 1. 各 category 记录数
category_counts = {}
for f in glob.glob(os.path.join(DATA_DIR, "*.jsonl")):
    name = os.path.basename(f).replace(".jsonl", "")
    if name.startswith("group_"):
        continue
    count = sum(1 for line in open(f, encoding="utf-8") if line.strip())
    category_counts[name] = count

with open(os.path.join(OUT_DIR, "category_counts.json"), "w") as f:
    json.dump(category_counts, f, ensure_ascii=False)
print("category_counts.json done:", category_counts)

# 2. sa 每日交易金额
daily = {}
sa_path = os.path.join(DATA_DIR, "sa.jsonl")
if os.path.exists(sa_path):
    for line in open(sa_path, encoding="utf-8"):
        line = line.strip()
        if not line:
            continue
        try:
            row = json.loads(line)
            val = row.get("value", "{}")
            if isinstance(val, str):
                val = json.loads(val)
            date = val.get("tran_date", "unknown")
            amt  = float(val.get("tran_amt", 0) or 0)
            daily[date] = daily.get(date, 0) + amt
        except:
            pass

dates = sorted(daily.keys())
with open(os.path.join(OUT_DIR, "sa_daily.json"), "w") as f:
    json.dump({"dates": dates, "amounts": [daily[d] for d in dates]}, f)
print("sa_daily.json done:", len(dates), "dates")
```

### 第二步：写一个把所有数据内嵌进去的 HTML

```bash
python3 ~/viz/build_html.py
```

**`~/viz/build_html.py`** 内容：

```python
#!/usr/bin/env python3
import json, os

OUT_DIR = os.path.expanduser("~/viz/static")

cat  = json.load(open(os.path.join(OUT_DIR, "category_counts.json")))
sa   = json.load(open(os.path.join(OUT_DIR, "sa_daily.json")))

html = """<!DOCTYPE html>
<html lang="zh-CN">
<head>
<meta charset="UTF-8">
<title>银行流式数据可视化</title>
<script src="https://cdn.jsdelivr.net/npm/echarts@5/dist/echarts.min.js"></script>
<style>
  body { font-family: sans-serif; background:#f0f2f5; margin:0; padding:24px; }
  h1   { color:#1a1a2e; }
  .card { background:white; border-radius:10px; padding:16px;
          box-shadow:0 2px 10px rgba(0,0,0,.08); margin-bottom:24px; }
  .chart { width:100%; height:380px; }
</style>
</head>
<body>
<h1>📊 银行流式数据可视化</h1>

<div class="card">
  <h2>各交易类别数据量</h2>
  <div id="c1" class="chart"></div>
</div>

<div class="card">
  <h2>储蓄账户 (sa) 每日交易金额</h2>
  <div id="c2" class="chart"></div>
</div>

<script>
var CAT_DATA = """ + json.dumps(cat, ensure_ascii=False) + """;
var SA_DATA  = """ + json.dumps(sa,  ensure_ascii=False) + """;

// 图1
(function(){
  var keys = Object.keys(CAT_DATA).sort();
  echarts.init(document.getElementById('c1')).setOption({
    tooltip: { trigger:'axis' },
    xAxis: { type:'category', data:keys, axisLabel:{rotate:30, fontSize:12} },
    yAxis: { type:'value', name:'记录数' },
    series:[{ type:'bar', data:keys.map(k=>CAT_DATA[k]),
              itemStyle:{color:'#5470c6'}, label:{show:true, position:'top'} }]
  });
})();

// 图2
(function(){
  echarts.init(document.getElementById('c2')).setOption({
    tooltip: { trigger:'axis' },
    xAxis: { type:'category', data:SA_DATA.dates, axisLabel:{rotate:45, fontSize:11} },
    yAxis: { type:'value', name:'金额(元)' },
    series:[{ type:'line', data:SA_DATA.amounts, smooth:true,
              itemStyle:{color:'#91cc75'}, areaStyle:{opacity:0.3} }]
  });
})();
</script>
</body>
</html>
"""

out = os.path.expanduser("~/viz/static/index.html")
with open(out, "w", encoding="utf-8") as f:
    f.write(html)
print("index.html written to", out)
```
