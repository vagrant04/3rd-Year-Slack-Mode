    public static void main(String[] args) throws Exception {

                "org.apache.kafka.common.serialization.StringDeserializer");

        props.put(SaslConfigs.*SASL_MECHANISM*, "PLAIN");

                "org.apache.kafka.common.security.plain.PlainLoginModule required username=\""
                        + username + "\" password=\"" + password + "\";");

        KafkaConsumer<String, String> consumer = new KafkaConsumer<>(props);
        consumer.subscribe(Collections.*singletonList*(topic));

        Runtime.*getRuntime*().addShutdownHook(new Thread(() -> {
            try {
                *closeWriters*();
                consumer.close();
                System.*out*.println("Consumer closed.");
            } catch (Exception e) {
                e.printStackTrace();
            }
        }));

        System.*out*.println("Start consuming topic: " + topic);
        System.*out*.println("Group ID: " + groupId);
        System.*out*.println("Output directory: " + outputDir);

        long count = 0;

        while (true) {
            ConsumerRecords<String, String> records = consumer.poll(Duration.*ofMillis*(1000));

            for (ConsumerRecord<String, String> record : records) {
                String category = *getHeader*(record, "category");
                String groupNo = *getHeader*(record, "groupNo");

                if (groupNo != null && !groupNo.trim().isEmpty()) {
                    *writeLine*(outputDir + "/group_" + groupNo + ".jsonl",
                            *toJsonLine*(record, "groupNo", groupNo));
                } else if (category != null && !category.trim().isEmpty()) {
                    *writeLine*(outputDir + "/" + category + ".jsonl",
                            *toJsonLine*(record, "category", category));
                } else {
                    *writeLine*(outputDir + "/unknown.jsonl",
                            *toJsonLine*(record, "unknown", "unknown"));
                }

                count++;

                if (count % 1000 == 0) {
                    *flushWriters*();
                    System.*out*.println("Consumed records: " + count);
                }
            }
        }
    }

    private static String getHeader(ConsumerRecord<String, String> record, String name) {
        Header header = record.headers().lastHeader(name);
        if (header == null || header.value() == null) {
            return null;
        }
        return new String(header.value(), StandardCharsets.*UTF_8*);
    }

    private static void writeLine(String filePath, String line) throws Exception {
        BufferedWriter writer = *WRITERS*.get(filePath);

        if (writer == null) {
            File file = new File(filePath);
            File parent = file.getParentFile();

            if (parent != null) {
                parent.mkdirs();
            }

            writer = new BufferedWriter(new OutputStreamWriter(
                    new FileOutputStream(file, true),
                    StandardCharsets.*UTF_8*
*            *));

            *WRITERS*.put(filePath, writer);
        }

        writer.write(line);
        writer.newLine();
    }

    private static String toJsonLine(
            ConsumerRecord<String, String> record,
            String headerName,
            String headerValue
    ) {
        return "{"
                + "\"topic\":\"" + *escapeJson*(record.topic()) + "\","
                + "\"partition\":" + record.partition() + ","
                + "\"offset\":" + record.offset() + ","
                + "\"timestamp\":" + record.timestamp() + ","
                + "\"" + *escapeJson*(headerName) + "\":\"" + *escapeJson*(headerValue) + "\","
                + "\"key\":\"" + *escapeJson*(record.key()) + "\","
                + "\"value\":\"" + *escapeJson*(record.value()) + "\""
                + "}";
    }

    private static String escapeJson(String text) {
        if (text == null) {
            return "";
        }

        return text
                .replace("\\", "\\\\")
                .replace("\"", "\\\"")
                .replace("\r", "\\r")
                .replace("\n", "\\n")
                .replace("\t", "\\t");
    }

    private static void flushWriters() throws Exception {
        for (BufferedWriter writer : *WRITERS*.values()) {
            writer.flush();
        }
    }

    private static void closeWriters() throws Exception {
        for (BufferedWriter writer : *WRITERS*.values()) {
            writer.flush();
            writer.close();
        }
        *WRITERS*.clear();
    }
}

package org.example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.apache.kafka.clients.producer.KafkaProducer;
import org.apache.kafka.clients.producer.ProducerRecord;

import java.io.BufferedReader;
import java.io.File;
import java.io.FileInputStream;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.util.Arrays;
import java.util.Comparator;
import java.util.Properties;

public class LocalReplayProducer {
    private static final ObjectMapper *MAPPER *= new ObjectMapper();

    public static void main(String[] args) throws Exception {
        String configPath = args.length > 0 ? args[0] : "producer.properties";

        Properties config = new Properties();
        config.load(new FileInputStream(configPath));

        String bootstrapServers = config.getProperty("bootstrap.servers");
        String topic = config.getProperty("topic", "bank_transaction");
        String inputDir = config.getProperty("input.dir", "data");
        long intervalMs = Long.*parseLong*(config.getProperty("send.interval.ms", "10"));

        Properties props = new Properties();
        props.put("bootstrap.servers", bootstrapServers);
        props.put("acks", config.getProperty("acks", "all"));
        props.put("retries", config.getProperty("retries", "3"));
        props.put("batch.size", config.getProperty("batch.size", "16384"));
        props.put("linger.ms", config.getProperty("linger.ms", "5"));
        props.put("buffer.memory", config.getProperty("buffer.memory", "33554432"));
        props.put("key.serializer", "org.apache.kafka.common.serialization.StringSerializer");
        props.put("value.serializer", "org.apache.kafka.common.serialization.StringSerializer");

        KafkaProducer<String, String> producer = new KafkaProducer<>(props);

        File dir = new File(inputDir);
        File[] files = dir.listFiles((file, name) ->
                name.endsWith(".jsonl")
                        && !name.startsWith("group_")
                        && !name.equals("unknown.jsonl")
        );

        if (files == null || files.length == 0) {
            System.*out*.println("No input jsonl files found in: " + inputDir);
            producer.close();
            return;
        }

        Arrays.*sort*(files, Comparator.*comparing*(File::getName));

        System.*out*.println("Start replaying data to Kafka.");
        System.*out*.println("Bootstrap servers: " + bootstrapServers);
        System.*out*.println("Topic: " + topic);
        System.*out*.println("Input directory: " + inputDir);

        long totalCount = 0;

        for (File file : files) {
            String category = *getCategoryFromFileName*(file.getName());
            long fileCount = 0;

            System.*out*.println("Sending file: " + file.getName() + ", category: " + category);

            try (BufferedReader reader = Files.*newBufferedReader*(file.toPath(), StandardCharsets.*UTF_8*)) {
                String line;

                while ((line = reader.readLine()) != null) {
                    if (line.trim().isEmpty()) {
                        continue;
                    }

                    JsonNode node = *MAPPER*.readTree(line);
                    String value = node.has("value") ? node.get("value").asText() : line;

                    ProducerRecord<String, String> record =
                            new ProducerRecord<>(topic, category, value);

                    record.headers().add(
                            "category",
                            category.getBytes(StandardCharsets.*UTF_8*)
                    );

                    producer.send(record);

                    totalCount++;
                    fileCount++;

                    if (totalCount % 1000 == 0) {
                        producer.flush();
                        System.*out*.println("Sent records: " + totalCount);
                    }

                    if (intervalMs > 0) {
                        Thread.*sleep*(intervalMs);
                    }
                }
            }

            producer.flush();
            System.*out*.println("Finished file: " + file.getName() + ", sent: " + fileCount);
        }

        producer.flush();
        producer.close();

        System.*out*.println("Replay finished. Total sent records: " + totalCount);
    }

    private static String getCategoryFromFileName(String fileName) {
        if (fileName.endsWith(".jsonl")) {
            return fileName.substring(0, fileName.length() - ".jsonl".length());
        }
        return fileName;
    }
}

                System.out.println("Total records processed: " + totalRecords.get());
                System.out.println("By category:");
                for (Map.Entry<String, AtomicLong> entry : categoryCounts.entrySet()) {
                    if (entry.getValue().get() > 0) {
                        System.out.println("  - " + entry.getKey() + ": " + entry.getValue().get());
                    }
                }
                // 取消所有任务
                try {
                    Runtime.getRuntime().exit(0);
                } catch (Exception e) {
                    e.printStackTrace();
                }
            }
        }, 10000, 10000, TimeUnit.MILLISECONDS);

        // 7. 执行任务
        try {
            env.execute("Flink-Kafka-ClickHouse-Transaction-ETL");
        } finally {
            idleChecker.shutdown();
        }
        
        System.out.println();
        System.out.println("========================================");
        System.out.println("Job completed successfully!");
        System.out.println("Final Statistics:");
        System.out.println("Total records processed: " + totalRecords.get());
        for (Map.Entry<String, AtomicLong> entry : categoryCounts.entrySet()) {
            if (entry.getValue().get() > 0) {
                System.out.println("  - " + entry.getKey() + ": " + entry.getValue().get());
            }
        }
        System.out.println("========================================");
    }
}

package com.bank;

import org.apache.flink.streaming.api.functions.sink.RichSinkFunction;

public class ClickHouseSink extends RichSinkFunction<String> {
    private ClickHouseConnection conn;
    private final String tableName;
    private final String category;
    private List<String> columnNames;
    
    private static final int BATCH_SIZE = 1000;  
    private static final long FLUSH_INTERVAL_MS = 30000; 
    private List<String[]> batchBuffer;
    private PreparedStatement batchStmt;
    private ScheduledExecutorService flushScheduler;
    private volatile long lastFlushTime = System.currentTimeMillis();

    private static final Map<String, AtomicLong> sinkCounts = new HashMap<>();
    private static final AtomicLong totalSuccess = new AtomicLong(0);
    private static final AtomicLong totalErrors = new AtomicLong(0);

    @Override
    public void open(Configuration parameters) throws Exception {
        super.open(parameters);
        String url = "jdbc:clickhouse://localhost:8123/dm";
        ClickHouseDataSource dataSource = new ClickHouseDataSource(url);
        conn = (ClickHouseConnection) dataSource.getConnection("default", "");
        
        this.columnNames = getColumnNames(category);
        this.batchBuffer = new ArrayList<>(BATCH_SIZE);
      
        StringBuilder columns = new StringBuilder();
        StringBuilder values = new StringBuilder();
        for (int i = 0; i < columnNames.size(); i++) {
            if (i > 0) {
                columns.append(",");
                values.append(",");
            }
            columns.append(columnNames.get(i));
            values.append("?");
        }
        
        String sql = String.format("INSERT INTO %s (%s) VALUES (%s)", tableName, columns, values);
        batchStmt = conn.prepareStatement(sql);
        
        System.out.println("[SINK INIT] " + category + " -> " + tableName + 
                          " (fields: " + columnNames.size() + ", batch_size: " + BATCH_SIZE + 
                          ", flush_interval: " + FLUSH_INTERVAL_MS + "ms)");
        
        // 启动定时刷新线程
        flushScheduler = Executors.newSingleThreadScheduledExecutor(r -> {
            Thread t = new Thread(r, "ClickHouseSink-Flush-" + category);
            t.setDaemon(true);
            return t;
        });
        flushScheduler.scheduleAtFixedRate(() -> {
            try {
                synchronized (ClickHouseSink.this) {
                    long idleTime = System.currentTimeMillis() - lastFlushTime;
                    if (idleTime >= FLUSH_INTERVAL_MS && batchBuffer != null && !batchBuffer.isEmpty()) {
                        System.out.println("[AUTO FLUSH] " + category + " idle for " + idleTime + "ms, flushing " + 
                                         batchBuffer.size() + " records");
                        executeBatch();
                    }
                }
            } catch (Exception e) {
                System.err.println("[AUTO FLUSH ERROR] " + category + ": " + e.getMessage());
            }
        }, FLUSH_INTERVAL_MS, FLUSH_INTERVAL_MS, TimeUnit.MILLISECONDS);
    }

    @Override
    public void invoke(String csvLine, Context context) throws Exception {
        try {
            
            String[] fields = csvLine.split(",");
            
            // 检查字段数量是否匹配
            if (fields.length != columnNames.size()) {
                totalErrors.incrementAndGet();
                System.err.println("[ERROR] Field count mismatch: category=" + category + 
                                 ", expected=" + columnNames.size() + 
                                 ", actual=" + fields.length);
                return;
            }
            
            
            synchronized (this) {
                batchBuffer.add(fields);
                
                
                if (batchBuffer.size() >= BATCH_SIZE) {
                    executeBatch();
                }
            }
            
        } catch (Exception e) {
            totalErrors.incrementAndGet();
            System.err.println("[ERROR] Failed to process record for " + category + ": " + e.getMessage());
            e.printStackTrace();
        }
    }
    
    private void executeBatch() throws Exception {
        if (batchBuffer.isEmpty()) return;
        
        int batchCount = batchBuffer.size();
        
        try {
           
            for (String[] fields : batchBuffer) {
                for (int i = 0; i < fields.length; i++) {
                    String value = fields[i].trim();
                    if (value.isEmpty() || value.equals("NULL") || value.equals("null")) {
                        batchStmt.setObject(i + 1, null);
                    } else {
                        batchStmt.setObject(i + 1, value);
                    }
                }
                batchStmt.addBatch();
            }
            
         
            int[] results = batchStmt.executeBatch();
            batchStmt.clearBatch();
            batchBuffer.clear();
            
            
            long count = sinkCounts.get(category).addAndGet(batchCount);
            totalSuccess.addAndGet(batchCount);
            long batches = totalBatches.incrementAndGet();
            lastFlushTime = System.currentTimeMillis();  // 更新最后刷新时间
            
            
            if (batches % 100 == 0) {
                System.out.println("[SINK BATCH] " + category + " (" + tableName + "): " + 
                                 count + " records in " + batches + " batches");
            }
            
        } catch (Exception e) {
            totalErrors.addAndGet(batchCount);
            System.err.println("[ERROR] Batch insert failed for " + category + ": " + e.getMessage());
            e.printStackTrace();
          
            batchStmt.clearBatch();
            batchBuffer.clear();
            throw e;
        }
    }

    // 关闭连接
    @Override
    public void close() throws Exception {
       
        if (flushScheduler != null) {
            flushScheduler.shutdown();
            try {
                flushScheduler.awaitTermination(5, TimeUnit.SECONDS);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }
        
       
        synchronized (this) {
            if (batchBuffer != null && !batchBuffer.isEmpty()) {
                System.out.println("[FINAL FLUSH] " + category + " flushing remaining " + batchBuffer.size() + " records");
                executeBatch();
            }
        }
        
        if (batchStmt != null) batchStmt.close();
        if (conn != null) conn.close();
        
        long count = sinkCounts.get(category).get();
        System.out.println("[SINK COMPLETE] " + category + " (" + tableName + "): " + 
                          count + " records processed");
        
        super.close();
    }

        System.out.println("========================================");
        System.out.println("Final Sink Statistics:");
        System.out.println("Total successful: " + totalSuccess.get());
        System.out.println("Total errors: " + totalErrors.get());
        System.out.println("By category:");
        for (Map.Entry<String, AtomicLong> entry : sinkCounts.entrySet()) {
            if (entry.getValue().get() > 0) {
                System.out.println("  - " + entry.getKey() + ": " + entry.getValue().get());
            }
        }
        System.out.println("========================================");
    }
